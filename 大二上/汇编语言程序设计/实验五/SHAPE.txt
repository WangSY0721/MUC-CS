.MODEL SMALL
.STACK 100H
.DATA

.CODE
START:
    MOV AX,@DATA
    MOV DS,AX

    MOV CX,0

OUTER_LOOP:
    PUSH CX
    MOV BX,CX
    INC BX

INNER_LOOP:
    PUSH BX
    MOV AX,BX
    DEC AX
    ADD AL,'0'
    MOV DL,AL
    MOV AH,02H
    INT 21H
    MOV DL,' '
    INT 21H

    POP BX
    DEC BX
    CMP BX,0
    JNZ INNER_LOOP

    MOV DL,13
    INT 21H
    MOV DL,10
    INT 21H

    POP CX
    INC CX
    CMP CX,9
    JBE OUTER_LOOP

END_PROGRAM:
    MOV AX,4C00H
    INT 21H

END START
