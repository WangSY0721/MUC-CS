.MODEL SMALL
.STACK 100H
.DATA
    STRING1 DB 'LIHUA',0
    STRING2 DB 'LIHUA',0
    MATCH_MSG DB 'MATCH$'
    NO_MATCH_MSG DB 'NO MATCH$'

.CODE
START:
    MOV AX,@DATA
    MOV DS,AX
    MOV SI,OFFSET STRING1
    MOV DI,OFFSET STRING2

COMPARE_LOOP:
    MOV AL,[SI]
    MOV BL,[DI]
    CMP AL,BL
    JNE NO_MATCH
    CMP AL,0
    JE MATCH
    INC SI
    INC DI
    JMP COMPARE_LOOP

MATCH:
    MOV DX,OFFSET MATCH_MSG
    JMP PRINT_MESSAGE

NO_MATCH:
    MOV DX,OFFSET NO_MATCH_MSG

PRINT_MESSAGE:
    MOV AH,9
    INT 21H

EXIT_PROGRAM:
    MOV AX,4C00H
    INT 21H

END START
